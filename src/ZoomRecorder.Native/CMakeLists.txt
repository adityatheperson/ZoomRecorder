cmake_minimum_required(VERSION 3.28)
project(ZoomRecorderNative LANGUAGES CXX)
enable_testing()
option(ZR_WITH_ZOOM "Build the licensed Zoom Meeting SDK adapter" OFF)

add_library(ZoomRecorder.Native SHARED src/api.cpp src/zoom/zoom_event_mapper.cpp src/media/audio_mixer.cpp src/media/capture_crop.cpp src/media/recording_readiness.cpp src/media/meeting_region_source.cpp src/media/wasapi_source.cpp src/media/mp4_writer.cpp src/media/recording_pipeline.cpp)
target_compile_features(ZoomRecorder.Native PRIVATE cxx_std_20)
target_include_directories(ZoomRecorder.Native PUBLIC include)
target_compile_definitions(ZoomRecorder.Native PRIVATE UNICODE _UNICODE)
target_link_libraries(ZoomRecorder.Native PRIVATE d3d11 windowsapp ole32 mfplat mfreadwrite mfuuid)

if(ZR_WITH_ZOOM)
  if(NOT DEFINED ENV{ZOOM_MEETING_SDK_DIR})
    message(FATAL_ERROR "ZOOM_MEETING_SDK_DIR must point to the licensed Zoom Meeting SDK for Windows")
  endif()
  set(ZOOM_SDK_DIR "$ENV{ZOOM_MEETING_SDK_DIR}")
  if(NOT EXISTS "${ZOOM_SDK_DIR}/x64/h/zoom_sdk.h" OR NOT EXISTS "${ZOOM_SDK_DIR}/x64/lib/sdk.lib")
    message(FATAL_ERROR "ZOOM_MEETING_SDK_DIR is missing x64/h/zoom_sdk.h or x64/lib/sdk.lib")
  endif()
  target_sources(ZoomRecorder.Native PRIVATE src/zoom/zoom_meeting_client.cpp)
  target_include_directories(ZoomRecorder.Native PRIVATE "${ZOOM_SDK_DIR}/x64/h")
  target_link_directories(ZoomRecorder.Native PRIVATE "${ZOOM_SDK_DIR}/x64/lib")
  target_link_libraries(ZoomRecorder.Native PRIVATE sdk)
  target_compile_definitions(ZoomRecorder.Native PRIVATE ZR_WITH_ZOOM=1)
endif()

add_executable(ZoomRecorder.Native.Tests ../../tests/ZoomRecorder.Native.Tests/api_tests.cpp ../../tests/ZoomRecorder.Native.Tests/zoom_event_mapper_tests.cpp ../../tests/ZoomRecorder.Native.Tests/media_tests.cpp ../../tests/ZoomRecorder.Native.Tests/mp4_writer_tests.cpp ../../tests/ZoomRecorder.Native.Tests/capture_crop_tests.cpp src/zoom/zoom_event_mapper.cpp src/media/audio_mixer.cpp src/media/capture_crop.cpp src/media/recording_readiness.cpp src/media/mp4_writer.cpp)
target_compile_features(ZoomRecorder.Native.Tests PRIVATE cxx_std_20)
target_include_directories(ZoomRecorder.Native.Tests PRIVATE src/zoom src/media)
target_link_libraries(ZoomRecorder.Native.Tests PRIVATE ZoomRecorder.Native d3d11 mfplat mfreadwrite mfuuid)
add_test(NAME native_api COMMAND ZoomRecorder.Native.Tests)
