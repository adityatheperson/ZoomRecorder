cmake_minimum_required(VERSION 3.28)
project(ZoomRecorderNative LANGUAGES CXX)
enable_testing()
add_library(ZoomRecorder.Native SHARED src/api.cpp src/media/aspect_fit.cpp src/media/audio_chunk_exporter.cpp src/media/audio_mixer.cpp src/media/audio_sample_timeline.cpp src/media/capture_crop.cpp src/media/recording_readiness.cpp src/media/meeting_window_watchdog.cpp src/media/meeting_region_source.cpp src/media/wasapi_source.cpp src/media/mp4_writer.cpp src/media/video_frame_normalizer.cpp src/media/recording_pipeline.cpp)
target_compile_features(ZoomRecorder.Native PRIVATE cxx_std_20)
target_include_directories(ZoomRecorder.Native PUBLIC include)
target_compile_definitions(ZoomRecorder.Native PRIVATE UNICODE _UNICODE)
target_link_libraries(ZoomRecorder.Native PRIVATE bcrypt d3d11 windowsapp ole32 mf mfplat mfreadwrite mfuuid)

add_executable(ZoomRecorder.Native.Tests ../../tests/ZoomRecorder.Native.Tests/api_tests.cpp ../../tests/ZoomRecorder.Native.Tests/media_tests.cpp ../../tests/ZoomRecorder.Native.Tests/mp4_writer_tests.cpp ../../tests/ZoomRecorder.Native.Tests/capture_crop_tests.cpp ../../tests/ZoomRecorder.Native.Tests/meeting_window_watchdog_tests.cpp ../../tests/ZoomRecorder.Native.Tests/audio_chunk_exporter_tests.cpp ../../tests/ZoomRecorder.Native.Tests/aspect_fit_tests.cpp ../../tests/ZoomRecorder.Native.Tests/video_frame_normalizer_tests.cpp src/media/aspect_fit.cpp src/media/audio_chunk_exporter.cpp src/media/audio_mixer.cpp src/media/audio_sample_timeline.cpp src/media/capture_crop.cpp src/media/recording_readiness.cpp src/media/meeting_window_watchdog.cpp src/media/mp4_writer.cpp src/media/video_frame_normalizer.cpp)
target_compile_features(ZoomRecorder.Native.Tests PRIVATE cxx_std_20)
target_include_directories(ZoomRecorder.Native.Tests PRIVATE src/media)
target_link_libraries(ZoomRecorder.Native.Tests PRIVATE ZoomRecorder.Native bcrypt d3d11 mf mfplat mfreadwrite mfuuid)
add_test(NAME native_api COMMAND ZoomRecorder.Native.Tests)
